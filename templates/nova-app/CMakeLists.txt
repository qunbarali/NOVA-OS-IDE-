cmake_minimum_required(VERSION 3.20)
project(nova_app_example C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(NOVA_APP_TARGET "rp2040" CACHE STRING "rp2040 or rp2350")
set(NOVA_APP_RAM_ORIGIN "" CACHE STRING "Reserved application RAM origin from the final NOVA linker map")
set(NOVA_APP_RAM_LENGTH "" CACHE STRING "Reserved application RAM length, for example 0x8000")
if(NOT NOVA_APP_RAM_ORIGIN OR NOT NOVA_APP_RAM_LENGTH)
    message(FATAL_ERROR "Set NOVA_APP_RAM_ORIGIN and NOVA_APP_RAM_LENGTH from the final NOVA firmware memory map. Do not guess these values.")
endif()
if(NOVA_APP_TARGET STREQUAL "rp2040")
    set(NOVA_CPU_FLAGS -mcpu=cortex-m0plus -mthumb)
elseif(NOVA_APP_TARGET STREQUAL "rp2350")
    set(NOVA_CPU_FLAGS -mcpu=cortex-m33 -mthumb)
else()
    message(FATAL_ERROR "NOVA_APP_TARGET must be rp2040 or rp2350.")
endif()

configure_file(nova_app.ld.in ${CMAKE_CURRENT_BINARY_DIR}/nova_app.ld @ONLY)
add_executable(nova_app src/main.c)
target_include_directories(nova_app PRIVATE include)
target_compile_options(nova_app PRIVATE ${NOVA_CPU_FLAGS} -Os -ffreestanding -fno-builtin -ffunction-sections -fdata-sections -fno-unwind-tables -fno-asynchronous-unwind-tables -Wall -Wextra -Werror)
target_link_options(nova_app PRIVATE ${NOVA_CPU_FLAGS} -nostdlib -Wl,-T,${CMAKE_CURRENT_BINARY_DIR}/nova_app.ld -Wl,-e,nova_app_main -Wl,--gc-sections -Wl,--build-id=none -Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/nova_app.map)
set_target_properties(nova_app PROPERTIES OUTPUT_NAME "nova_app" SUFFIX ".elf")
