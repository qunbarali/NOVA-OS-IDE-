cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
if(NOT PICO_SDK_PATH)
    message(FATAL_ERROR "Set PICO_SDK_PATH to your Raspberry Pi Pico SDK directory.")
endif()
include(pico_sdk_import.cmake)
project(nova_hello_app C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(nova_hello main.c)
target_link_libraries(nova_hello pico_stdlib)
pico_enable_stdio_usb(nova_hello 1)
pico_enable_stdio_uart(nova_hello 0)
pico_add_extra_outputs(nova_hello)
